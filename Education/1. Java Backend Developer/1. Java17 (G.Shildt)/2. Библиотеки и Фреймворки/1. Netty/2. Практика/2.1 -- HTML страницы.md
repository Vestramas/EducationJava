<!-- ====== JavaScript ====== -->  
<script>  
    const API_URL = '/api/users';  
  
    // --- Загрузка списка пользователей ---  
    function loadUsers() {  
        const container = document.getElementById('usersContainer');  
        container.innerHTML = '<p style="color: #64748b;">Загрузка...</p>';  
  
        fetch(API_URL)  
            .then(response => {  
                if (!response.ok) {  
                    throw new Error(`HTTP ${response.status}`);  
                }  
                return response.json();  
            })  
            .then(users => {  
                container.innerHTML = '';  
                if (users.length === 0) {  
                    container.innerHTML = '<p style="color: #64748b;">Пользователей пока нет</p>';  
                    return;  
                }  
                users.forEach(user => {  
                    const card = document.createElement('div');  
                    card.className = 'user-card';  
                    card.dataset.id = user.id;  
                    card.innerHTML = `  
                            <h3>${escapeHtml(user.name)}</h3>  
                            <p><i class="fas fa-envelope"></i> ${escapeHtml(user.email)}</p>  
                            <div class="card-actions">                                <button class="btn-edit" onclick="openEditModal(${user.id}, '${escapeHtml(user.name)}', '${escapeHtml(user.email)}')">  
                                    <i class="fas fa-edit"></i> Обновить                                </button>                                <button class="btn-delete" onclick="deleteUser(${user.id})">  
                                    <i class="fas fa-trash"></i> Удалить                                </button>                            </div>                        `;  
                    container.appendChild(card);  
                });  
            })  
            .catch(error => {  
                console.error('Ошибка загрузки:', error);  
                container.innerHTML = '<p style="color: #ef4444;">Ошибка загрузки пользователей</p>';  
            });  
    }  
  
    // --- Защита от XSS ---  
    function escapeHtml(text) {  
        if (!text) return '';  
        const div = document.createElement('div');  
        div.textContent = text;  
        return div.innerHTML;  
    }  
  
    // --- Открыть модалку для добавления ---  
    function openAddModal() {  
        document.getElementById('editUserId').value = '';  
        document.getElementById('userName').value = '';  
        document.getElementById('userEmail').value = '';  
        document.getElementById('modalTitle').textContent = 'Добавление пользователя';  
        document.getElementById('saveUserBtn').textContent = 'Создать';  
        document.getElementById('userModal').classList.add('active');  
    }  
  
    // --- Открыть модалку для редактирования ---  
    function openEditModal(id, name, email) {  
        document.getElementById('editUserId').value = id;  
        document.getElementById('userName').value = name;  
        document.getElementById('userEmail').value = email;  
        document.getElementById('modalTitle').textContent = 'Редактирование пользователя';  
        document.getElementById('saveUserBtn').textContent = 'Обновить';  
        document.getElementById('userModal').classList.add('active');  
    }  
  
    // --- Закрыть модалку ---  
    function closeModal() {  
        document.getElementById('userModal').classList.remove('active');  
    }  
  
    // --- Сохранить пользователя (POST или PUT) ---  
    function saveUser() {  
        const name = document.getElementById('userName').value.trim();  
        const email = document.getElementById('userEmail').value.trim();  
  
        if (!name || !email) {  
            alert('Пожалуйста, заполните все поля.');  
            return;  
        }  
  
        const userId = document.getElementById('editUserId').value;  
        const isEditing = userId !== '';  
        const url = isEditing ? `${API_URL}?id=${userId}` : API_URL;  
        const method = isEditing ? 'PUT' : 'POST';  
        const body = { name, email };  
  
        fetch(url, {  
            method: method,  
            headers: { 'Content-Type': 'application/json' },  
            body: JSON.stringify(body)  
        })  
            .then(response => {  
                if (!response.ok) {  
                    throw new Error(`HTTP ${response.status}`);  
                }  
                closeModal();  
                loadUsers();  
            })  
            .catch(error => {  
                console.error('Ошибка сохранения:', error);  
                alert('Не удалось сохранить пользователя');  
            });  
    }  
  
    // --- Удалить пользователя ---  
    function deleteUser(id) {  
        if (!confirm('Вы уверены, что хотите удалить этого пользователя?')) {  
            return;  
        }  
  
        fetch(`${API_URL}?id=${id}`, {  
            method: 'DELETE'  
        })  
            .then(response => {  
                if (!response.ok) {  
                    throw new Error(`HTTP ${response.status}`);  
                }  
                loadUsers();  
            })  
            .catch(error => {  
                console.error('Ошибка удаления:', error);  
                alert('Не удалось удалить пользователя');  
            });  
    }  
  
    // --- Закрытие модалки по Esc ---  
    document.addEventListener('keydown', function(e) {  
        if (e.key === 'Escape') {  
            closeModal();  
        }  
    });

    document.getElementById('userModal').addEventListener('click', function(e) {  
        if (e.target === this) {  
            closeModal();

<!-- ====== JavaScript ====== -->  
<script>  
    let socket = null;  
    let isConnected = false;  
  
    const WS_URL = 'ws://localhost:8080/chat-ws';  
  
    const connectBtn = document.getElementById('connectBtn');  
    const disconnectBtn = document.getElementById('disconnectBtn');  
    const sendBtn = document.getElementById('sendBtn');  
    const messageInput = document.getElementById('messageInput');  
    const chatMessages = document.getElementById('chatMessages');  
    const statusDot = document.getElementById('statusDot');  
    const statusText = document.getElementById('statusText');  
  
    // --- Обновление статуса ---  
    function setStatus(state, message) {  
        statusDot.className = 'status-dot';  
        if (state === 'connected') {  
            statusDot.classList.add('connected');  
            statusText.textContent = message || 'Подключен';  
        } else if (state === 'error') {  
            statusDot.classList.add('error');  
            statusText.textContent = message || 'Ошибка';  
        } else {  
            statusDot.classList.add('offline');  
            statusText.textContent = message || 'Не подключен';  
        }  
    }  
  
    // --- Обновление UI ---  
    function updateUI(connected) {  
        isConnected = connected;  
        connectBtn.disabled = connected;  
        disconnectBtn.disabled = !connected;  
        messageInput.disabled = !connected;  
        sendBtn.disabled = !connected;  
  
        if (connected) {  
            setStatus('connected', 'Подключен');  
        } else {  
            setStatus('offline', 'Не подключен');  
        }  
    }  
  
    // --- Добавление сообщения в чат ---  
    function addMessage(text, type) {  
        // Удаляем пустое состояние, если оно есть  
        const emptyState = chatMessages.querySelector('.empty-state');  
        if (emptyState) {  
            emptyState.remove();  
        }  
  
        const msg = document.createElement('div');  
        msg.className = `chat-message ${type}`;  
        msg.textContent = text;  
        chatMessages.appendChild(msg);  
        chatMessages.scrollTop = chatMessages.scrollHeight;  
    }  
  
    // --- Подключение к WebSocket ---  
    function connect() {  
        if (socket && isConnected) {  
            return;  
        }  
  
        try {  
            socket = new WebSocket(WS_URL);  
  
            socket.onopen = function() {  
                console.log('WebSocket соединение установлено');  
                updateUI(true);  
                addMessage('Соединение установлено', 'system');  
            };  
  
            socket.onmessage = function(event) {  
                console.log('Получено сообщение:', event.data);  
                addMessage(event.data, 'received');  
            };  
  
            socket.onerror = function(error) {  
                console.error('WebSocket ошибка:', error);  
                setStatus('error', 'Ошибка соединения');  
                addMessage('Ошибка соединения', 'system');  
            };  
  
            socket.onclose = function(event) {  
                console.log('WebSocket соединение закрыто', event.code, event.reason);  
                updateUI(false);  
                if (!event.wasClean) {  
                    addMessage('Соединение разорвано', 'system');  
                } else {  
                    addMessage('Соединение закрыто', 'system');  
                }  
                socket = null;  
            };  
  
        } catch (e) {  
            console.error('Ошибка создания WebSocket:', e);  
            setStatus('error', 'Ошибка подключения');  
            addMessage('Не удалось подключиться к серверу', 'system');  
        }  
    }  
  
    // --- Отключение ---  
    function disconnect() {  
        if (socket) {  
            socket.close(1000, 'Отключено пользователем');  
        } else {  
            updateUI(false);  
            addMessage('Отключено', 'system');  
        }  
    }  
  
    // --- Отправка сообщения ---  
    function sendMessage() {  
        const text = messageInput.value.trim();  
        if (!text || !socket || socket.readyState !== WebSocket.OPEN) {  
            return;  
        }  
  
        socket.send(text);  
        addMessage(text, 'sent');  
        messageInput.value = '';  
        messageInput.focus();  
    }  
  
    // --- Отправка по Enter ---  
    messageInput.addEventListener('keydown', function(e) {  
        if (e.key === 'Enter') {  
            e.preventDefault();  
            sendMessage();  
        }  
    });  
  
    // --- Закрытие сокета при закрытии страницы ---  
    window.addEventListener('beforeunload', function() {  
        if (socket) {  
            socket.close(1000, 'Страница закрыта');  
        }  
    });  
  
    // --- Инициализация UI ---  
    updateUI(false);  
</script>
